Headers: Authorization: Bearer YOUR_WSUGGET_KEY

// Wsugget Cloud Storage File Upload Logic
async function uploadToWsugget(fileObject) {
    const formData = new FormData();
    formData.append('file', fileObject);

    const response = await fetch('https://api.wsugget.vercel.app/storage/upload', {
        method: 'POST',
        headers: {
            'Authorization': 'Bearer YOUR_WSUGGET_KEY'
        },
        body: formData
    });

    const result = await response.json();
    if(result.success) {
        console.log(`Cloud URI Matrix Generated: ${result.fileUrl}`); // users/username/filename
    }
}
import requests

def upload_file_matrix(file_path, api_key):
    url = "https://api.wsugget.vercel.app/storage/upload"
    headers = {"Authorization": f"Bearer {api_key}"}
    files = {'file': open(file_path, 'rb')}
    
    response = requests.post(url, headers=headers, files=files)
    if response.status_code == 200:
        print("Storage target path initialized:", response.json().get("fileUrl"))
