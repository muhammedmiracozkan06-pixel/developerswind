window.addEventListener('DOMContentLoaded', () => {
    const urlParams = new URLSearchParams(window.location.search);
    const status = urlParams.get('status');
    if (status === 'success') {
        const username = urlParams.get('username');
        const email = urlParams.get('email');
        const token = urlParams.get('token');
        console.log(`Identity verified: ${username} (${email})`);
        localStorage.setItem('wmail_session', JSON.stringify({ username, email, token }));
        window.history.replaceState({}, document.title, window.location.pathname);
        alert(`Welcome authorized entity: ${username}! Cluster synchronized.`);
    }
});
from flask import Flask, request, jsonify
app = Flask(__name__)

@app.route('/callback', methods=['GET'])
def wmail_callback_handler():
    status = request.args.get('status')
    if status == 'success':
        username = request.args.get('username')
        email = request.args.get('email')
        token = request.args.get('token')
        return jsonify({
            "authenticated": True,
            "user": username,
            "email": email,
            "token_verified": token
        }), 200
    return jsonify({"error": "Unauthorized handshake matrix rejected"}), 401

if __name__ == '__main__':
    app.run(port=5000)

Cloud HTML5 PC

Cloud HTML5 PC is a fully virtualized desktop environment written entirely in HTML5, CSS, and JavaScript. It runs inside your browser and leverages your local hardware resources — CPU, RAM, GPU — through modern Web APIs (WebGL, WebAssembly, and WebRTC). No plugins, no native binaries. Just a browser and an internet connection.

This is not a remote desktop viewer; it's a native‑in‑browser computing sandbox that executes logic on your machine while synchronizing state with the Wind cloud mesh. Your session is ephemeral, secure, and automatically garbage‑collected when you close the tab.

  • Uses your CPU & GPU
  • RAM & storage are browser‑managed
  • No server‑side execution — privacy first
  • State sync with Wind mesh
  • Ephemeral session — close tab = clean slate
  • Sub‑millisecond UI response
